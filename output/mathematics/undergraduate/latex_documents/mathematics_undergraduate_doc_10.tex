\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item In xyz-space, what are the coordinates of the point on the plane 2 x + y + 3 z = 3 that is closest to the origin?
\begin{enumerate}[label=(\alph*)]
    \item (0, 0, 1)
    \item (3/7, 3/14, 9/14)
    \item (7/15, 8/15, 1/15)
    \item (5/6, 1/3, 1/3)
\end{enumerate}
\item Statement 1 | For any two groups G and G', there exists a homomorphism of G into G'. Statement 2 | Every homomorphism is a one-to-one map.
\begin{enumerate}[label=(\alph*)]
    \item True, True
    \item False, False
    \item True, False
    \item False, True
\end{enumerate}
\item What is the units digit in the standard decimal expansion of the number $7^25$?
\begin{enumerate}[label=(\alph*)]
    \item 1
    \item 3
    \item 5
    \item 7
\end{enumerate}
\item (1+$i)^10$ =
\begin{enumerate}[label=(\alph*)]
    \item 1
    \item i
    \item 32
    \item 32 i
\end{enumerate}
\item The map x -\textgreater{} $axa^2$ of a group G into itself is a homomorphism if and only if
\begin{enumerate}[label=(\alph*)]
    \item G is abelian
    \item G = \{e\}
    \item $a^3$ = e
    \item $a^2$ = a
\end{enumerate}
\end{enumerate}
