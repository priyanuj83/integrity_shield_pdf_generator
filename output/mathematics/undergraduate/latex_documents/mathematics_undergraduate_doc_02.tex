\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Statement 1 | Every ideal in a ring is a subring of the ring. Statement 2 | Every subring of every ring is an ideal of the ring.
\begin{enumerate}[label=(\alph*)]
    \item True, True
    \item False, False
    \item True, False
    \item False, True
\end{enumerate}
\item Sofia and Tess will each randomly choose one of the 10 integers from 1 to 10. What is the probability that neither integer chosen will be the square of the other?
\begin{enumerate}[label=(\alph*)]
    \item 0.64
    \item 0.72
    \item 0.81
    \item 0.95
\end{enumerate}
\item The polynomial $x^4$ + 4 can be factored into linear factors in Z\_5[x]. Find this factorization.
\begin{enumerate}[label=(\alph*)]
    \item (x - 2)(x + 2)(x - 1)(x + 1)
    \item (x+$1)^4$
    \item (x-1)(x+$1)^3$
    \item (x-$1)^3(x$+1)
\end{enumerate}
\item Some group (G, 0) is known to be abelian. Then which one of the following is TRUE for G?
\begin{enumerate}[label=(\alph*)]
    \item g = g^-1 for every g in G
    \item g = $g^2$ for every g in G
    \item (g o $h)^2$ = $g^2$ o $h^2$ for every g,h in G
    \item G is of finite order
\end{enumerate}
\item Suppose today is Wednesday. What day of the week will it be $10^(10$^(10)) days from now?
\begin{enumerate}[label=(\alph*)]
    \item Sunday
    \item Monday
    \item Tuesday
    \item Wednesday
\end{enumerate}
\end{enumerate}
