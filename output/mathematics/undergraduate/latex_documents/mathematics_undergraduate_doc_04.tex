\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Let A be the set of all ordered pairs of integers (m, n) such that 7 m + 12 n = 22. What is the greatest negative number in the set B = \{m + n : (m, n) \ensuremath{\in} A\}?
\begin{enumerate}[label=(\alph*)]
    \item -5
    \item -4
    \item -3
    \item -2
\end{enumerate}
\item For how many positive integers k does the ordinary decimal representation of the integer k! end in exactly 99 zeros?
\begin{enumerate}[label=(\alph*)]
    \item None
    \item One
    \item Four
    \item Five
\end{enumerate}
\item Find the maximum possible order for some element of Z\_8 x Z\_10 x Z\_24.
\begin{enumerate}[label=(\alph*)]
    \item 8
    \item 120
    \item 240
    \item 24
\end{enumerate}
\item Find the generator for the finite field Z\_11.
\begin{enumerate}[label=(\alph*)]
    \item 1
    \item 2
    \item 3
    \item 4
\end{enumerate}
\item Let (Z\_\{10\} , +, x) be the ring of integers modulo 10, and let S be the subset of Z\_\{10\} represented by \{0, 2, 4, 6, 8\}. Which of the following statements is FALSE?
\begin{enumerate}[label=(\alph*)]
    \item (S, +, x) is closed under addition modulo 10.
    \item (S, +, x) is closed under multiplication modulo 10.
    \item (S, +, x) has an identity under addition modulo 10.
    \item (S, +, x) has no identity under multiplication modulo 10.
\end{enumerate}
\end{enumerate}

\end{enumerate}
