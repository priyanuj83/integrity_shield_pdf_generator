\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Determine whether the polynomial in Z[x] satisfies an Eisenstein criterion for irreducibility over Q. 8 $x^3$ + 6 $x^2$ - 9 x + 24
\begin{enumerate}[label=(\alph*)]
    \item Yes, with p=2.
    \item Yes, with p=3.
    \item Yes, with p=5.
    \item No.
\end{enumerate}
\item If a polynomial f(x) over the real numbers has the complex numbers 2 + i and 1 - i as roots, then f(x) could be
\begin{enumerate}[label=(\alph*)]
    \item $x^3$ + 5 $x^2$ + 4 x + 1
    \item $x^4$ - 6 $x^3$ + 15 $x^2$ - 18 x + 10
    \item $x^3$ - $x^2$ + 4 x + 1
    \item $x^4$ + 7 $x^2$ + 10
\end{enumerate}
\item Statement 1 | Every integral domain with characteristic 0 is infinite. Statement 2 | Every integral domain with prime characteristic is finite.
\begin{enumerate}[label=(\alph*)]
    \item True, True
    \item False, False
    \item True, False
    \item False, True
\end{enumerate}
\item Let I != A != -I, where I is the identity matrix and A is a real 2 x 2 matrix. If A = $A^($-1), then the trace of A is
\begin{enumerate}[label=(\alph*)]
    \item 2
    \item 1
    \item 0
    \item -1
\end{enumerate}
\item Let f(x,y) = $x^2$ - 2 xy + $y^3$ for all real x and y. Which of the following is true?
\begin{enumerate}[label=(\alph*)]
    \item f has all of its relative extrema on the line x = y.
    \item f has all of its relative extrema on the parabola x = $y^2$.
    \item f has a relative minimum at (0, 0).
    \item f has an absolute minimum at (2/3, 2/3).
\end{enumerate}
\end{enumerate}
