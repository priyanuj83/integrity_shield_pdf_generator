\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item If (G, .) is a group such that (ab)^-1 = a^-1 b^-1, for all a, b in G, then G is a/an
\begin{enumerate}[label=(\alph*)]
    \item commutative semi group
    \item abelian group
    \item non-abelian group
    \item None of these
\end{enumerate}
\item If A = (1, 2, 3, 4). Let \textasciitilde{}= \{(1, 2), (1, 3), (4, 2)\}. Then \textasciitilde{} is
\begin{enumerate}[label=(\alph*)]
    \item not anti-symmetric
    \item transitive
    \item reflexive
    \item symmetric
\end{enumerate}
\item Find all zeros in the indicated finite field of the given polynomial with coefficients in that field. $x^5$ + 3 $x^3$ + $x^2$ + 2 x in Z\_5
\begin{enumerate}[label=(\alph*)]
    \item 0
    \item 1
    \item 0,1
    \item 0,4
\end{enumerate}
\item A group G in which ($ab)^2$ = $a^2$ $b^2$ for all a, b in G is necessarily
\begin{enumerate}[label=(\alph*)]
    \item finite
    \item cyclic
    \item of order two
    \item abelian
\end{enumerate}
\item Find the maximum possible order for an element of S\_n for n = 7.
\begin{enumerate}[label=(\alph*)]
    \item 6
    \item 12
    \item 30
    \item 105
\end{enumerate}
\end{enumerate}

\section*{True / False (10 marks)}
\textit{Answer True or False}
\begin{enumerate}[label=\arabic*.]
\setcounter{enumi}{5}
\item True or False: If f(1 + x) = f(x) for all real x and f(5) = 11, then f(15/2) also equals 11.
\item True or False: Every permutation can be expressed as a product of disjoint cycles.
\item True or False: The circle described by the equation $x^2$ + $y^2$ = 1 intersects the parabola $x^2$ = y + 4 at the greatest number of points.
\item True or False: The product of (2,3) and (3,5) in the ring Z\_5 x Z\_9 is (3,1).
\item True or False: A function is a permutation if and only if it is one-to-one and onto, which is not always the case.
\end{enumerate}
