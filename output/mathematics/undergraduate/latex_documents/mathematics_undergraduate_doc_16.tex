\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item If (G, .) is a group such that (ab)^-1 = a^-1 b^-1, for all a, b in G, then G is a/an
\begin{enumerate}[label=(\alph*)]
    \item commutative semi group
    \item abelian group
    \item non-abelian group
    \item None of these
\end{enumerate}
\item Find the order of the factor group Z\_6/\textless{}3\textgreater{}.
\begin{enumerate}[label=(\alph*)]
    \item 2
    \item 3
    \item 6
    \item 12
\end{enumerate}
\item If A = \{1, 2, 3\} then relation S = \{(1, 1), (2, 2)\} is
\begin{enumerate}[label=(\alph*)]
    \item symmetric only
    \item anti-symmetric only
    \item both symmetric and anti-symmetric
    \item an equivalence relation
\end{enumerate}
\item Let A be the set of all ordered pairs of integers (m, n) such that 7 m + 12 n = 22. What is the greatest negative number in the set B = \{m + n : (m, n) \ensuremath{\in} A\}?
\begin{enumerate}[label=(\alph*)]
    \item -5
    \item -4
    \item -3
    \item -2
\end{enumerate}
\item In the xy-plane, the curve with parametric equations x = cos t and y = sin t, 0 \textless{}= t \textless{}= pi, has length
\begin{enumerate}[label=(\alph*)]
    \item 3
    \item pi
    \item 3 pi
    \item 3/2
\end{enumerate}
\end{enumerate}
