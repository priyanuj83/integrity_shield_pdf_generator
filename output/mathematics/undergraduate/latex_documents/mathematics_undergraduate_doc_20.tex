\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Statement 1 | For any two groups G and G', there exists a homomorphism of G into G'. Statement 2 | Every homomorphism is a one-to-one map.
\begin{enumerate}[label=(\alph*)]
    \item True, True
    \item False, False
    \item True, False
    \item False, True
\end{enumerate}
\item If A is a subset of the real line R and A contains each rational number, which of the following must be true?
\begin{enumerate}[label=(\alph*)]
    \item If A is open, then A = R.
    \item If A is closed, then A = R.
    \item If A is uncountable, then A = R.
    \item If A is uncountable, then A is open.
\end{enumerate}
\item The element (4, 2) of Z\_12 x Z\_8 has order
\begin{enumerate}[label=(\alph*)]
    \item 4
    \item 8
    \item 12
    \item 6
\end{enumerate}
\item Suppose that f(1 + x) = f(x) for all real x. If f is a polynomial and f(5) = 11, then f(15/2)
\begin{enumerate}[label=(\alph*)]
    \item -11
    \item 0
    \item 11
    \item 33/2
\end{enumerate}
\item For which of the following rings is it possible for the product of two nonzero elements to be zero?
\begin{enumerate}[label=(\alph*)]
    \item The ring of complex numbers
    \item The ring of integers modulo 11
    \item The ring of continuous real-valued functions on [0, 1]
    \item The ring \{a + b * sqrt(2) : a and b are rational numbers\}
\end{enumerate}
\end{enumerate}
