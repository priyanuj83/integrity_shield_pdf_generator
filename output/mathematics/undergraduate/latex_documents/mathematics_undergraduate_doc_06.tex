\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item For what value of b is the line y = 10 x tangent to the curve y = $e^(bx$) at some point in the xy-plane?
\begin{enumerate}[label=(\alph*)]
    \item 10/e
    \item 10
    \item 10 e
    \item $e^10$
\end{enumerate}
\item How many structurally distinct Abelian groups have order 72?
\begin{enumerate}[label=(\alph*)]
    \item 4
    \item 6
    \item 8
    \item 9
\end{enumerate}
\item Which of the following is an equation of the line tangent to the graph of y = x + $e^x$ at x = 0 ?
\begin{enumerate}[label=(\alph*)]
    \item y = 2 x + 1
    \item y = 2 x
    \item y = x + 2
    \item y = x + 1
\end{enumerate}
\item The element (4, 2) of Z\_12 x Z\_8 has order
\begin{enumerate}[label=(\alph*)]
    \item 4
    \item 8
    \item 12
    \item 6
\end{enumerate}
\item Find all c in Z\_3 such that Z\_3[x]/($x^3$ + $cx^2$ + 1) is a field.
\begin{enumerate}[label=(\alph*)]
    \item 0
    \item 2
    \item 1
    \item 3
\end{enumerate}
\end{enumerate}
