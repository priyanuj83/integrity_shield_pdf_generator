\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Statement 1 | The external direct product of cyclic groups is cyclic. Statement 2 | The external direct product of D\_3 and D\_4 is isomorphic to D\_12.
\begin{enumerate}[label=(\alph*)]
    \item True, True
    \item False, False
    \item True, False
    \item False, True
\end{enumerate}
\item Statement 1 | If a R is an integral domain, then R[x] is an integral domain. Statement 2 | If R is a ring and f(x) and g(x) are in R[x], then deg (f(x)g(x)) = deg f(x) + deg g(x).
\begin{enumerate}[label=(\alph*)]
    \item True, True
    \item False, False
    \item True, False
    \item False, True
\end{enumerate}
\item Let (Z\_\{10\} , +, x) be the ring of integers modulo 10, and let S be the subset of Z\_\{10\} represented by \{0, 2, 4, 6, 8\}. Which of the following statements is FALSE?
\begin{enumerate}[label=(\alph*)]
    \item (S, +, x) is closed under addition modulo 10.
    \item (S, +, x) is closed under multiplication modulo 10.
    \item (S, +, x) has an identity under addition modulo 10.
    \item (S, +, x) has no identity under multiplication modulo 10.
\end{enumerate}
\item If the finite group G contains a subgroup of order seven but no element (other than the identity) is its own inverse, then the order of G could be
\begin{enumerate}[label=(\alph*)]
    \item 27
    \item 28
    \item 35
    \item 37
\end{enumerate}
\item Find the maximum possible order for some element of Z\_4 x Z\_6.
\begin{enumerate}[label=(\alph*)]
    \item 4
    \item 6
    \item 12
    \item 24
\end{enumerate}
\end{enumerate}
