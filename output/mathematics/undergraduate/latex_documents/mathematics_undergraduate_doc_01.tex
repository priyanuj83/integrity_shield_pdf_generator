\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Determine whether the polynomial in Z[x] satisfies an Eisenstein criterion for irreducibility over Q. $x^2$ - 12
\begin{enumerate}[label=(\alph*)]
    \item Yes, with p=2.
    \item Yes, with p=3.
    \item Yes, with p=5.
    \item No.
\end{enumerate}
\item Statement 1 | Every nonzero free abelian group has an infinite number of bases. Statement 2 | Every free abelian group of rank at least 2 has an infinite number of bases.
\begin{enumerate}[label=(\alph*)]
    \item True, True
    \item False, False
    \item True, False
    \item False, True
\end{enumerate}
\item Suppose A, B, and C are statements such that C is true if exactly one of A and B is true. If C is false, which of the following statements must be true?
\begin{enumerate}[label=(\alph*)]
    \item If A is true, then B is false.
    \item If A is false, then B is false.
    \item If A is false, then B is true.
    \item Both A and B are true.
\end{enumerate}
\item Let f(x) be a function whose graph passes through the origin. If f(2 n) = $n^2$ + f[2(n - 1)] for every integer n, what is the value of f(8)?
\begin{enumerate}[label=(\alph*)]
    \item 24
    \item 30
    \item 32
    \item 36
\end{enumerate}
\item Find the characteristic of the ring Z x Z.
\begin{enumerate}[label=(\alph*)]
    \item 0
    \item 3
    \item 12
    \item 30
\end{enumerate}
\end{enumerate}
