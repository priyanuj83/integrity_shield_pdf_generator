\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Statement 1 | Every integral domain has a field of quotients. Statement 2 | A polynomial of degree n over a ring can have at most n zeros counting multiplicity.
\begin{enumerate}[label=(\alph*)]
    \item True, True
    \item False, False
    \item True, False
    \item False, True
\end{enumerate}
\item Statement 1 | The homomorphic image of a cyclic group is cyclic. Statement 2 | The homomorphic image of an Abelian group is Abelian.
\begin{enumerate}[label=(\alph*)]
    \item True, True
    \item False, False
    \item True, False
    \item False, True
\end{enumerate}
\item For which value of n are there exactly two abelian groups of order n up to isomorphism?
\begin{enumerate}[label=(\alph*)]
    \item 4
    \item 7
    \item 8
    \item 12
\end{enumerate}
\item Let f(x, y) = $x^3$ + $y^3$ + 3 xy for all real x and y. Then there exist distinct points P and Q such that f has a
\begin{enumerate}[label=(\alph*)]
    \item local maximum at P and at Q
    \item saddle point at P and at Q
    \item local maximum at P and a saddle point at Q
    \item local minimum at P and a saddle point at Q
\end{enumerate}
\item Find the maximum possible order for an element of S\_n for n = 6.
\begin{enumerate}[label=(\alph*)]
    \item 6
    \item 12
    \item 30
    \item 105
\end{enumerate}
\end{enumerate}
