\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Of the cans of soup in Rolando's cupboard, 1/2 are tomato and 2/5 are chicken noodle. What fraction of the cans of soup in Rolando's cupboard are either tomato or chicken noodle?
\begin{enumerate}[label=(\alph*)]
    \item 2/3
    \item 10-Sep
    \item 5-Mar
    \item 6/11
    \item 5/8
\end{enumerate}
\item The value of the digit 5 in 24,513 is how many times the value of the digit 5 in 357?
\begin{enumerate}[label=(\alph*)]
    \item 0
    \item 50
    \item 100
    \item 100,000
    \item 5,000
\end{enumerate}
\item If a series circuit has a capacitor of $C=0.8 \times 10^{-6} \mathrm{~F}$ and an inductor of $L=0.2 \mathrm{H}$, find the resistance $R$ so that the circuit is critically damped.
\begin{enumerate}[label=(\alph*)]
    \item 500 $\Omega$
    \item 400 $\Omega$
    \item 3000 $\Omega$
    \item 1250 $\Omega$
    \item 1000 $\Omega$
\end{enumerate}
\item Let V be the space spanned by functions cos(2 x) and sin(2 x). Find the determinant of the linear transformation D(f) = f' from V to V.
\begin{enumerate}[label=(\alph*)]
    \item 0
    \item -1
    \item 1
    \item 2
    \item -2
\end{enumerate}
\item How many ways are there to partition a set of 5 elements into 3 non-empty cycles?
\begin{enumerate}[label=(\alph*)]
    \item 45
    \item 10
    \item 15
    \item 40
    \item 50
\end{enumerate}
\end{enumerate}
