\begin{enumerate}[label=\arabic*.]
\item Which of the following sets has the greatest cardinality?
\begin{enumerate}[label=(\alph*)]

    \item R
    \item The set of all infinite subsets of R
    \item The set of all functions from Z to Z
    \item The set of all functions from \{0, 1\} to R

    \item 1/25
    \item 1/30
    \item 3/75
    \item 4/100
\end{enumerate}

    \item \$0.60

\end{enumerate}
