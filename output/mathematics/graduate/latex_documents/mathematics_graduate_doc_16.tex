\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Let g(x) be the inverse of f(x) = x + cos(x). What is g'(1)?
\begin{enumerate}[label=(\alph*)]
    \item sqrt(2)
    \item -2
    \item 1
    \item 0
    \item -1/2
\end{enumerate}
\item Find the degree for the given field extension Q(sqrt(2), sqrt(3), sqrt(18)) over Q.
\begin{enumerate}[label=(\alph*)]
    \item 3
    \item 10
    \item 0
    \item 12
    \item 5
\end{enumerate}
\item In a group of 10 scores, the largest score is increased by 40 points. What will happen to the mean?
\begin{enumerate}[label=(\alph*)]
    \item It will remain the same.
    \item It will increase by 30 points.
    \item It will increase by 10 points.
    \item It will decrease by 10 points.
    \item It will remain unchanged if the other scores also increase by 40 points.
\end{enumerate}
\item What is the second number in the row of Pascal's triangle that has 43 numbers?
\begin{enumerate}[label=(\alph*)]
    \item 42
    \item 1
    \item 2
    \item 40
    \item 22
\end{enumerate}
\item Let $P_5(x)$ be the fifth-degree Taylor polynomial approximation for f(x)=sin(x), centered at x=0. What is the Lagrange error of the polynomial approximation to sin(1)?.
\begin{enumerate}[label=(\alph*)]
    \item 0.02098
    \item 0.00198
    \item 0.000198
    \item 0.01998
    \item 0.0198
\end{enumerate}
\end{enumerate}
