\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Let A be the set of all ordered pairs of integers (m, n) such that 7 m + 12 n = 22. What is the greatest negative number in the set B = \{m + n : (m, n) \ensuremath{\in} A\}?
\begin{enumerate}[label=(\alph*)]
    \item -5
    \item 0
    \item -3
    \item -7
    \item -4
\end{enumerate}
\item A model of a park was built on a scale of 1.5 centimeters to 50 meters. If the distance between two trees in the park is 150 meters, what is this distance on the model?
\begin{enumerate}[label=(\alph*)]
    \item 6 centimeters
    \item 150 centimeters
    \item 0.5 centimeter
    \item 9 centimeters
    \item 225 centimeters
\end{enumerate}
\item Let V and W be 4-dimensional subspaces of a 7-dimensional vector space X. Which of the following CANNOT be the dimension of the subspace V intersect W?
\begin{enumerate}[label=(\alph*)]
    \item 0
    \item 1
    \item 9
    \item 8
    \item 2
\end{enumerate}
\item Solve 13 over 4 = x over 7.
\begin{enumerate}[label=(\alph*)]
    \item 22.25
    \item 23.25
    \item 25.25
    \item 23.75
    \item 21.75
\end{enumerate}
\item If all the values of a data set are the same, all of the following must equal zero except for which one?
\begin{enumerate}[label=(\alph*)]
    \item Mean
    \item Standard deviation
    \item Variance
    \item Interquartile range
    \item Quartile deviation
\end{enumerate}
\end{enumerate}

\section*{True / False (10 marks)}
\textit{Answer True or False}
\begin{enumerate}[label=\arabic*.]
\setcounter{enumi}{5}
\item True or False: The sum of the second partial derivatives of u with respect to x and y equals 2.0.
\item True or False: The arc length of y = (1/$4)x^4$ over the interval [1,2] calculated using the Trapezoidal Rule T\_5 is 4.322.
\item True or False: The fraction \( \frac{4}{\sqrt{108}+2\sqrt{12}+2\sqrt{27}} \) simplifies to \( \frac{1}{2\sqrt{3}} \).
\item True or False: Every group of order 159 is cyclic, as it is the product of two distinct primes and thus must have a non-cyclic structure.
\item True or False: The limit of 3 n(a\_$n)^3$ as n approaches infinity is 1.0.
\end{enumerate}
