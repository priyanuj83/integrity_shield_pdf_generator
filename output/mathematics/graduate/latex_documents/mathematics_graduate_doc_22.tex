\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Keiko spent the day bird watching and counted 34 more birds in the morning than in the afternoon. If she counted a total of 76 birds, how many birds did she count in the afternoon?
\begin{enumerate}[label=(\alph*)]
    \item 84 birds
    \item 45 birds
    \item 21 birds
    \item 34 birds
    \item 42 birds
\end{enumerate}
\item If a freight train travels at a speed of 20 miles per hour for 6 hours, how far will it travel?
\begin{enumerate}[label=(\alph*)]
    \item 90 miles
    \item 120 miles
    \item 150 miles
    \item 140 miles
    \item 26 miles
\end{enumerate}
\item You and three friends go to a concert. The total cost for four tickets is \$112. Write and solve an equation to find the cost of one ticket t.
\begin{enumerate}[label=(\alph*)]
    \item 4 - t = 112; -\$108
    \item 4 t = 112; \$28
    \item t over 4 = 112; \$448
    \item t + 4 = 112; \$108
    \item t over 4 = 112; \$28
\end{enumerate}
\item what is the value of \ensuremath{\int}\_$a^b$ \ensuremath{\frac{dx}}\{\ensuremath{\sqrt{(x-a)(b-x)}}\}? Round the answer to the thousands decimal.
\begin{enumerate}[label=(\alph*)]
    \item 0.6931471
    \item 3.7320508
    \item 2.3025851
    \item 6.2831853
    \item 1.4142135
\end{enumerate}
\item What is the order of the element 5 in U\_8?
\begin{enumerate}[label=(\alph*)]
    \item 3
    \item 4
    \item 1
    \item 5
    \item 9
\end{enumerate}
\end{enumerate}
