\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item If x - 3 = 6, what is the value of x?
\begin{enumerate}[label=(\alph*)]
    \item 0
    \item 4
    \item 3
    \item 2
    \item 9
\end{enumerate}
\item Let f(x,y) = $x^2$ - 2 xy + $y^3$ for all real x and y. Which of the following is true?
\begin{enumerate}[label=(\alph*)]
    \item f has all of its relative extrema on the line y = $x^2$.
    \item f has all of its relative extrema on the line x = y.
    \item f has all of its relative extrema on the line x = 2 y.
    \item f has a relative maximum at (0, 0).
    \item f has an absolute maximum at (2/3, 2/3).
\end{enumerate}
\item For the regression line, which of the following statements about residuals is true?
\begin{enumerate}[label=(\alph*)]
    \item The sum of the residuals is always one.
    \item If the correlation is 0, there will be a distinct pattern in the residual plot.
    \item The larger the residuals, the better the linear model.
    \item The residuals are always negative.
    \item If the correlation is 1, there will be a distinct pattern in the residual plot.
\end{enumerate}
\item The polynomial $x^3 - Ax + 15$ has three real roots. Two of these roots sum to 5. What is |A|?
\begin{enumerate}[label=(\alph*)]
    \item 19.0
    \item 17.0
    \item 25.0
    \item 23.0
    \item 18.0
\end{enumerate}
\item Find 13 over 14 + 7 over 14.
\begin{enumerate}[label=(\alph*)]
    \item 20 over 28
    \item 2 and 3 over 14
    \item 1 and 3 over 7
    \item 2 and 1 over 14
    \item 3 and 2 over 7
\end{enumerate}
\end{enumerate}
