\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Statement 1 | For n \textgreater{} 1, the set \{1,2, ..., n-1\} is a group under multiplication modulo n. Statement 2 | There is an integer x such that 63 x mod 100 = 1.
\begin{enumerate}[label=(\alph*)]
    \item False, True
    \item False, False and x is less than 100
    \item False, True and x is an odd number
    \item True, True and x is greater than 100
    \item False, True and x is a prime number
\end{enumerate}
\item Both A, B are n-by-n matrices with rank(A)=n, rank(A*B)=0. What is rank(B)?
\begin{enumerate}[label=(\alph*)]
    \item 0.0
    \item Undefined
    \item 2
    \item n-1
    \item n+1
\end{enumerate}
\item A relative maximum value of the function y = (ln x)/x is
\begin{enumerate}[label=(\alph*)]
    \item e/2
    \item 1
    \item $e^2$
    \item 2/e
    \item 1/2
\end{enumerate}
\item Is there exist a holomorphic function $f$ on the unit disk $B(0,1)$ (boundary excluded) such that $f(B(0,1))=C$? Here C is the complex space. Return 1 for yes and 0 for no.
\begin{enumerate}[label=(\alph*)]
    \item 2.5
    \item -2.0
    \item 3.0
    \item 2.0
    \item 1.5
\end{enumerate}
\item What is ($6^83$ + $8^83$) mod 49?
\begin{enumerate}[label=(\alph*)]
    \item 42
    \item 14
    \item 35
    \item 21
    \item 49
\end{enumerate}
\end{enumerate}
