\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item What is the least perfect square with 3 different prime factors?
\begin{enumerate}[label=(\alph*)]
    \item 900
    \item 441
    \item 300
    \item 484
    \item 529
\end{enumerate}
\item Let M be a 5 x 5 real matrix. Exactly four of the following five conditions on M are equivalent to each other. Which of the five conditions is equivalent to NONE of the other four?
\begin{enumerate}[label=(\alph*)]
    \item The column vectors of M form a basis for $R^5$.
    \item For any two distinct column vectors u and v of M, the set \{u, v\} is linearly independent.
    \item The trace of M is nonzero.
    \item The eigenvalues of M are all nonzero.
    \item The homogeneous system Mx = 0 has only the trivial solution.
\end{enumerate}
\item Find all positive values of $c$ so that the inequality $x^2-6 x+c<0$ has real solutions for $x$. Express your answer in interval notation.
\begin{enumerate}[label=(\alph*)]
    \item (0, 0)
    \item (6, 0)
    \item (9, 16)
    \item (0, 9)
    \item (3, 6)
\end{enumerate}
\item Which pair of expressions below are equivalent?
\begin{enumerate}[label=(\alph*)]
    \item 2 x + 2 x and 2 $x^2$
    \item x+x+x+x and $x^4$
    \item 3 x + 5 x and 15 x
    \item 4(2 x - 6) and 8 x - 24
    \item 3 x + 3 x + 3 x and 3 $x^3$
\end{enumerate}
\item What is the probability that a randomly selected integer in the set $\{1,2,3,\ldots,100\}$\$ is divisible by 2 and not divisible by 3? Express your answer as a common fraction.
\begin{enumerate}[label=(\alph*)]
    \item \ensuremath{\frac{17}{31}}
    \item \ensuremath{\frac{33}{66}}
    \item \ensuremath{\frac{33}{100}}
    \item \ensuremath{\frac{17}{50}}
    \item \ensuremath{\frac{1}{6}}
\end{enumerate}
\end{enumerate}
