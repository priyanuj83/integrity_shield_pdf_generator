\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Mai earns $5.50 per hour at her after-school job. How many hours does she have to work to earn $132?
\begin{enumerate}[label=(\alph*)]
    \item 16
    \item 20
    \item 24
    \item 28
\end{enumerate}
\item How many square units are in the region satisfying the inequalities $y \ge |x|$ and $y \le -|x|+3$? Express your answer as a decimal.
\begin{enumerate}[label=(\alph*)]
    \item 4.5
    \item 2.25
    \item 6.6
    \item 3.3
\end{enumerate}
\item How many positive 4-digit integers have only odd digits?
\begin{enumerate}[label=(\alph*)]
    \item 100
    \item 999
    \item 625
    \item 500
\end{enumerate}
\item Define a new measurement as the difference between the 60 th and 40 th percentile scores in a population. This measurement will give information concerning
\begin{enumerate}[label=(\alph*)]
    \item central tendency.
    \item variability.
    \item symmetry.
    \item skewness.
\end{enumerate}
\item What is the smallest prime whose digits sum to $19$?
\begin{enumerate}[label=(\alph*)]
    \item 5
    \item 11
    \item 17
    \item 199
\end{enumerate}
\end{enumerate}
