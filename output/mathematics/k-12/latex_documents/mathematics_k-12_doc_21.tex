\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item What is the greatest common divisor of $2^{1001}-1$ and $2^{1012}-1$?
\begin{enumerate}[label=(\alph*)]
    \item 1
    \item 2049
    \item 2048
    \item 2047
\end{enumerate}
\item If x and y are directly proportional and x = 3 when y = 8, what is the value of x when y = 13?
\begin{enumerate}[label=(\alph*)]
    \item 0.615
    \item 4.875
    \item 15
    \item 34.667
\end{enumerate}
\item Which of the following points is NOT a solution to both the inequalities y \textgreater{} 9 x - 8 and y \textless{} - x + 8?
\begin{enumerate}[label=(\alph*)]
    \item (- 2, 15)
    \item (-1, 5)
    \item (0, 0)
    \item (1, 5)
\end{enumerate}
\item A hypothesis test is set up so that P(rejecting H$_{0}$ when H$_{0}$ is true) = 0.05 and P(failing to reject H$_{0}$ when H$_{0}$ is false) = 0.26. What is the power of the test?
\begin{enumerate}[label=(\alph*)]
    \item 0.26
    \item 0.05
    \item 0.95
    \item 0.74
\end{enumerate}
\item Find the value of the expression -16 + 13 + (-33).
\begin{enumerate}[label=(\alph*)]
    \item 4
    \item 62
    \item 36
    \item -36
\end{enumerate}
\end{enumerate}
