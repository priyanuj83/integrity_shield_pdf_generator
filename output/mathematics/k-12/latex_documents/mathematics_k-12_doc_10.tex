\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item What is the greatest common factor of 36 and 90?
\begin{enumerate}[label=(\alph*)]
    \item 6
    \item 18
    \item 36
    \item 180
\end{enumerate}
\item Given that the sample has a standard deviation of zero, which of the following is a true statement?
\begin{enumerate}[label=(\alph*)]
    \item The standard deviation of the population is also zero.
    \item The sample mean and sample median are equal.
    \item The sample may have outliers.
    \item The population has a symmetric distribution.
\end{enumerate}
\item What is the least possible positive integer-value of $n$ such that $\sqrt{18\cdot n \cdot 34}$ is an integer?
\begin{enumerate}[label=(\alph*)]
    \item 34
    \item 2
    \item 17
    \item 3
\end{enumerate}
\item Let $M=4 a^2 - 2 b^2 +a$. Let $j$ be the value of $M$ when $a=5$ and $b=3$, and let $k$ be the value of $M$ when $a=-1$ and $b=4$. Calculate $j+2 k$.
\begin{enumerate}[label=(\alph*)]
    \item 29
    \item 28
    \item 27
    \item 26
\end{enumerate}
\item What is the sum of all integer solutions to $|n| < |n-3| < 9$?
\begin{enumerate}[label=(\alph*)]
    \item -9
    \item -14
    \item 14
    \item 9
\end{enumerate}
\end{enumerate}
