\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Write the ratio of 96 runners to 216 swimmers in simplest form.
\begin{enumerate}[label=(\alph*)]
    \item 96 over 216
    \item 16 over 36
    \item 4 over 9
    \item 9 over 4
\end{enumerate}
\item Jess scored 18 points during her last basketball game. Each basket she made was worth 2 points. How many baskets did she make?
\begin{enumerate}[label=(\alph*)]
    \item 20
    \item 16
    \item 9
    \item 8
\end{enumerate}
\item Find the number that makes the statement true: 48 m = Box mm.
\begin{enumerate}[label=(\alph*)]
    \item 48,000
    \item 480
    \item 4,800
    \item 5
\end{enumerate}
\item Which number written in exponential notation is equivalent to 343?
\begin{enumerate}[label=(\alph*)]
    \item $3^7$
    \item $7^3$
    \item $7^49$
    \item $294^49$
\end{enumerate}
\item Emil wants to find out the most popular football team at a game between the home team and the visiting team. Which of the following methods will give him the most accurate results?
\begin{enumerate}[label=(\alph*)]
    \item surveying the cheerleaders for the home team
    \item surveying people wearing hats for the visiting team
    \item surveying a group of people standing in line for tickets
    \item surveying people who do not live in the home team's city
\end{enumerate}
\end{enumerate}
