\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]

    \item 77
    \item 78
    \item 79

    \item 42.25
    \item 32.54
    \item 38.44

    \item 12
    \item 22
    \item 840
    \item 240
\end{enumerate}
\item What is the shortest distance from the origin to the circle defined by $x^2-24 x +y^2+10 y +160=0$?

    \item 16
    \item 24
    \item 12
\end{enumerate}
\item Evaluate -2(x - 3) for x = 2.

    \item -4
    \item -2
    \item 10
    \item 2
\end{enumerate}
\end{enumerate}
