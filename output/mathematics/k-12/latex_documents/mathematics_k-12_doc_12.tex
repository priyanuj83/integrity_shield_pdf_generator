\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item The practice range at a golf course charges \$4.00 for a bucket of 40 golf balls. At this rate, how much will a bucket of 100 golf balls cost?
\begin{enumerate}[label=(\alph*)]
    \item \$10.00
    \item \$10.50
    \item \$13.50
    \item \$16.00
\end{enumerate}
\item Find the median in the set of data 23, 13, 18, 29, 32, 25.
\begin{enumerate}[label=(\alph*)]
    \item 18
    \item 24
    \item 25
    \item 29
\end{enumerate}
\item Which expression is equivalent to 5(4 x + 3) - 2 x?
\begin{enumerate}[label=(\alph*)]
    \item 18 x+15
    \item 18 x+3
    \item 7 x+8
    \item 2 x+8
\end{enumerate}
\item What is the value of the expression 2(3($4^2$ + 1)) - $2^3$?
\begin{enumerate}[label=(\alph*)]
    \item 156
    \item 110
    \item 94
    \item 48
\end{enumerate}
\item Which group of numbers is in order from least to greatest?
\begin{enumerate}[label=(\alph*)]
    \item 0.25 1.6 1.0
    \item 1.0 0.25 1.6
    \item 0.25 1.0 1.6
    \item 1.6 1.0 0.25
\end{enumerate}
\end{enumerate}
