\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Which ratio can form a proportion with 6 over 9?
\begin{enumerate}[label=(\alph*)]
    \item 8 over 12
    \item 9 over 6
    \item 2 over 6
    \item 24 over 27
\end{enumerate}
\item Find the sum of the primes between 100 and 200, inclusive, that are 1 or 2 more than a perfect square.
\begin{enumerate}[label=(\alph*)]
    \item 298
    \item 126
    \item 592
    \item 396
\end{enumerate}
\item Five thousand dollars compounded annually at an $x\%$ interest rate takes six years to double. At the same interest rate, how many years will it take $\$300$ to grow to $\$9600$?
\begin{enumerate}[label=(\alph*)]
    \item 12
    \item 1
    \item 30
    \item 5
\end{enumerate}
\item Which fraction shows the ratio of 3 dogs to 5 dogs?
\begin{enumerate}[label=(\alph*)]
    \item 5 over 3
    \item 3 over 8
    \item 2 over 5
    \item 3 over 5
\end{enumerate}
\item An ant crawls straight from $(-4,6)$ to the origin, and then continues straight on to $(4,3)$. How far does it travel?
\begin{enumerate}[label=(\alph*)]
    \item 16
    \item 2\ensuremath{\sqrt{2}}
    \item 5
    \item 5+2\ensuremath{\sqrt{13}}
\end{enumerate}
\end{enumerate}
