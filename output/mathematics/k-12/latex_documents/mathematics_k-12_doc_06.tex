\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Restate the expression 4(3 + 2) using the Distributive Property. Do not simplify.
\begin{enumerate}[label=(\alph*)]
    \item 4(3) + 2
    \item 4(3) + 4(2)
    \item 20
    \item 3 + 4(2)
\end{enumerate}
\item Which statementis true?
\begin{enumerate}[label=(\alph*)]
    \item The product of 5 x 2 is even because both of the factors are even.
    \item The product of 4 x 4 is odd because both of the factors are even.
    \item The productof 2 x 7 is even because both of the factors are odd.
    \item The product of 5 x 3 is odd because both of the factors are odd.
\end{enumerate}
\item Identify the algebraic expression that matches the word phrase: 4 times as many cans as Tom collected.
\begin{enumerate}[label=(\alph*)]
    \item 4 t
    \item t over 4
    \item t - 4
    \item t + 4
\end{enumerate}
\item How many positive and negative integers is $12$ a multiple of?
\begin{enumerate}[label=(\alph*)]
    \item 1
    \item 4
    \item 12
    \item 3
\end{enumerate}
\item Find 1 over 6 + 1 over 8.
\begin{enumerate}[label=(\alph*)]
    \item 7 over 24
    \item 2 over 14
    \item 1 over 7
    \item 1 over 4
\end{enumerate}
\end{enumerate}
