\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item The value of the digit 5 in 24,513 is how many times the value of the digit 5 in 357?
\begin{enumerate}[label=(\alph*)]
    \item 10
    \item 100
    \item 1,000
    \item 10,000
\end{enumerate}
\item Which expression is equivalent to 5 x 9?
\begin{enumerate}[label=(\alph*)]
    \item (5 x 4) x (6 x 5)
    \item (5 x 5) + (5 x 4)
    \item (5 x 5) + (5 x 9)
    \item (5 x 9) x (6 x 9)
\end{enumerate}
\item Which among the following would result in the narrowest confidence interval?
\begin{enumerate}[label=(\alph*)]
    \item Small sample size and 95\% confidence
    \item Small sample size and 99\% confidence
    \item Large sample size and 95\% confidence
    \item Large sample size and 99\% confidence
\end{enumerate}
\item A number's prime factors are 2, 5, 7, 13, and 31. Which of the following must be a factor of the number?
\begin{enumerate}[label=(\alph*)]
    \item 4
    \item 6
    \item 10
    \item 15
\end{enumerate}
\item What is the product of the greatest even prime number and the least odd prime number?
\begin{enumerate}[label=(\alph*)]
    \item 6
    \item 9
    \item 12
    \item 14
\end{enumerate}
\end{enumerate}
