\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item The polynomial which results from the expansion of $(x^2+5 x+6)^2+(px+q)(x^3+7 x^2+3 x)$ has degree $2$. Find $p+q$.
\begin{enumerate}[label=(\alph*)]
    \item -2
    \item -4
    \item -1
    \item 1
\end{enumerate}
\item How many numbers are in the list $25, 26, 27, \ldots, 99, 100 ?$
\begin{enumerate}[label=(\alph*)]
    \item 75
    \item 76
    \item 22
    \item 23
\end{enumerate}
\item The length of a rectangle is twice its width. Given the length of the diagonal is $5\sqrt{5}$, find the area of the rectangle.
\begin{enumerate}[label=(\alph*)]
    \item 2500
    \item 2
    \item 50
    \item 25
\end{enumerate}
\item Find the remainder when $25^1059$ is divided by 23.
\begin{enumerate}[label=(\alph*)]
    \item 11
    \item 13
    \item 8
    \item 5
\end{enumerate}
\item The sum $1^2 + 2^2 + 3^2 + 4^2 + \cdots + n^2 = n(n+1)(2 n+1) \div 6$. What is the value of $21^2 + 22^2 + \cdots + 40^2$?
\begin{enumerate}[label=(\alph*)]
    \item 41
    \item 22140
    \item 19270
    \item 2870
\end{enumerate}
\end{enumerate}
