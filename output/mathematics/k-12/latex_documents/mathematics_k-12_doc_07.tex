\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item There are 230 calories in 4 ounces of a type of ice cream. How many calories are in 6 ounces of that ice cream?
\begin{enumerate}[label=(\alph*)]
    \item 232
    \item 236
    \item 345
    \item 460
\end{enumerate}
\item A zoo has 15 toucans and 60 parrots. What is the ratio of the number of toucans to the number of parrots at the zoo?
\begin{enumerate}[label=(\alph*)]
    \item 1:04
    \item 1:05
    \item 4:01
    \item 4:05
\end{enumerate}
\item Balloons come in packages of 8. If you need 42 balloons, how many packages should you buy?
\begin{enumerate}[label=(\alph*)]
    \item 5
    \item 6
    \item 8
    \item 42
\end{enumerate}
\item Suppose two events, E and F, have nonzero probabilities p and q, respectively. Which of the following is impossible?
\begin{enumerate}[label=(\alph*)]
    \item p + q \textgreater{} 1
    \item p - q \textless{} 0
    \item p /q \textgreater{} 1
    \item E and F are both independent and mutually exclusive.
\end{enumerate}
\item Let $p$, $q$, and $r$ be constants. One solution to the equation $(x-p)(x-q) = (r-p)(r-q)$ is $x=r$. Find the other solution in terms of $p$, $q$, and $r$.
\begin{enumerate}[label=(\alph*)]
    \item pqr
    \item pq-r
    \item p-q-r
    \item p+q-r
\end{enumerate}
\end{enumerate}
