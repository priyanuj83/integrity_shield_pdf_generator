\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item What is the most reasonable estimate of the length of a city's swimming pool?
\begin{enumerate}[label=(\alph*)]
    \item 1 meter
    \item 25 meters
    \item 1 kilometer
    \item 25 kilometers
\end{enumerate}
\item The hypotenuse of a right triangle measures 10 inches and one angle is $45^{\circ}$. What is the number of square inches in the area of the triangle?
\begin{enumerate}[label=(\alph*)]
    \item 10
    \item 45
    \item 125
    \item 25
\end{enumerate}
\item For the equation: 3 x - 4(x - 2) + 6 x - 8 = 0, find the value of x.
\begin{enumerate}[label=(\alph*)]
    \item -4
    \item 0
    \item 2
    \item 4
\end{enumerate}
\item In how many ways can 5 students be selected from a group of 6 students?
\begin{enumerate}[label=(\alph*)]
    \item 12
    \item 3
    \item 5
    \item 6
\end{enumerate}
\item What is the domain of the function $f(x)=\frac{x+6}{\sqrt{x^2-3 x-4}}$?
\begin{enumerate}[label=(\alph*)]
    \item (-4,4)
    \item [0,4]
    \item (-inf, -1) U (4, inf)
    \item (-inf, -1) U (-1, 4) U (4, inf)
\end{enumerate}
\end{enumerate}
