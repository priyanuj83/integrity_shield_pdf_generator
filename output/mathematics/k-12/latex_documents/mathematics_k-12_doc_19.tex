\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item On a Cartesian coordinate plane, points $(1,2)$ and $(7, 4)$ are opposite vertices of a square. What is the area of the square?
\begin{enumerate}[label=(\alph*)]
    \item 20
    \item 34
    \item 40
    \item \ensuremath{\sqrt{40}}
\end{enumerate}
\item If we take sin x and change it to 3 sin 2 x, what will happen to the domain and range of the function?
\begin{enumerate}[label=(\alph*)]
    \item Domain and range remain the same
    \item Domain is double; range remains the same
    \item Domain is tripled; range is doubled
    \item Domain is the same; range is tripled
\end{enumerate}
\item Write 8 and 2 over 4 as an improper fraction in simplest form.
\begin{enumerate}[label=(\alph*)]
    \item 17 over 2
    \item 34 over 4
    \item 17 over 4
    \item 19 over 2
\end{enumerate}
\item Lucy is counting by 2 s. She starts with the number 2 and stops at the number 50. Which number would Lucy not count?
\begin{enumerate}[label=(\alph*)]
    \item 11
    \item 22
    \item 34
    \item 48
\end{enumerate}
\item If$_{50}$\% of a number is 20, what is 75\% of the number?
\begin{enumerate}[label=(\alph*)]
    \item 8
    \item 15
    \item 30
    \item 45
\end{enumerate}
\end{enumerate}
