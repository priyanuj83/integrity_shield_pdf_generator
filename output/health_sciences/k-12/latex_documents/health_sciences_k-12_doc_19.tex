\begin{enumerate}[label=(\alph*)]

    \item Type 2 obesity related diabetes remains a very rare problem

\begin{enumerate}[label=(\alph*)]

\begin{enumerate}[label=(\alph*)]

    \item Living on a working farm will help reducing the development of atopic diseases
    \item Particular care must be taken during breast feeding in high risk infants
    \item Living on a working farm will increase the development of atopic diseases
\end{enumerate}
\item Bone mineral accretion
\begin{enumerate}[label=(\alph*)]

    \item Is particularly high after the menopause in women
\end{enumerate}
