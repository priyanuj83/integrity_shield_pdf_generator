    \item 2.2 g
    \item 4.0 g

\begin{enumerate}[label=(\alph*)]

    \item MgS
    \item NaCl
    \item LiCl
    \item NaF

    \item 0.02 mg \%

    \item 13.00128 amu
    \item 13.00987 amu
    \item 13.00335 amu
    \item 13.00739 amu
\end{enumerate}

    \item 150 K
    \item 75 K
    \item 110 K
