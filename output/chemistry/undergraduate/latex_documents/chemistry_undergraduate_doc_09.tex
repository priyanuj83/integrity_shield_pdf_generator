\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item 23 Na has an NMR frequency of 198.52 MHz in a 750 MHz NMR spectrometer. What is its magnetogyric ratio?
\begin{enumerate}[label=(\alph*)]
    \item 101.1 x $10^7$ T-1 s-1
    \item 26.75 x $10^7$ T-1 s-1

    \item 0.9377 x $10^7$ T-1 s-1

\item Which of the following statements most accurately explains why the T$_{1}$ of a nucleus is sometimes longer than its T$_{2}$?
\begin{enumerate}[label=(\alph*)]
    \item T$_{1}$, unlike T$_{2}$, is sensitive to very low-frequency molecular motions.
    \item T$_{2}$, unlike T$_{1}$, is sensitive to very low-frequency molecular motions.

    \item T$_{2}$, unlike T$_{1}$, is sensitive to molecular motions at the Larmor frequency.

    \item II only
    \item III only
    \item I and III only
\end{enumerate}

    \item X-rays

\end{enumerate}
\end{enumerate}
