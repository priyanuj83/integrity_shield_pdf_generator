\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Calculate the Larmor frequency for a proton in a magnetic field of 1 T.
\begin{enumerate}[label=(\alph*)]
    \item 23.56 GHz
    \item 42.58 MHz
    \item 74.34 kHz
    \item 13.93 MHz
\end{enumerate}
\item Calculate the magnetic field responsible for the polarization of 2.5 x 10^-6 for 13 C at 298 K.
\begin{enumerate}[label=(\alph*)]
    \item 0.5 T
    \item 1.2 T
    \item 2.9 T
    \item 100 T
\end{enumerate}
\item When the following equation is balanced, which of the following is true? \_\_ MnO 4- + \_\_ I- + \_\_ H+ \textless{}-\textgreater{} \_\_ Mn$_{2}$+ + \_\_ IO 3- + \_\_ H$_{2}$O
\begin{enumerate}[label=(\alph*)]
    \item The I- : IO 3- ratio is 3:1.
    \item The MnO 4- : I- ratio is 6:5.
    \item The MnO 4- : Mn$_{2}$+ ratio is 3:1.
    \item The H+ : I- ratio is 2:1.
\end{enumerate}
\item The 13 C spectrum of which isomer of C$_{6}$H$_{14}$ has lines with three distinct chemical shifts?
\begin{enumerate}[label=(\alph*)]
    \item hexane
    \item 2-methylpentane
    \item 3-methylpentane
    \item 2,3-dimethylbutane
\end{enumerate}
\item Of the following ionic substances, which has the greatest lattice enthalpy?
\begin{enumerate}[label=(\alph*)]
    \item MgO
    \item MgS
    \item NaF
    \item NaCl
\end{enumerate}
\end{enumerate}
