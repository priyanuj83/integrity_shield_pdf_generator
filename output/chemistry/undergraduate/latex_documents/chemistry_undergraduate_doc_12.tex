\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Considering 0.1 M aqueous solutions of each of the following, which solution has the lowest pH?
\begin{enumerate}[label=(\alph*)]
    \item Na$_{2}$CO$_{3}$
    \item Na$_{3}$PO$_{4}$
    \item Na$_{2}$S
    \item NaCl
\end{enumerate}
\item Of the following ions, which has the smallest radius?
\begin{enumerate}[label=(\alph*)]
    \item K+
    \item Ca$_{2}$+
    \item Sc$_{3}$+
    \item Rb+
\end{enumerate}
\item Which of the following is an n-type semiconductor?
\begin{enumerate}[label=(\alph*)]
    \item Silicon
    \item Diamond
    \item Silicon carbide
    \item Arsenic-doped silicon
\end{enumerate}
\item Which of the following is always true of a spontaneous process?
\begin{enumerate}[label=(\alph*)]
    \item The process is exothermic.
    \item The process does not involve any work.
    \item The entropy of the system increases.
    \item The total entropy of the system plus surroundings increases.
\end{enumerate}
\item Calculate the Q-factor for an X-band EPR cavity with a resonator bandwidth of 1.58 MHz.
\begin{enumerate}[label=(\alph*)]
    \item Q = 1012
    \item Q = 2012
    \item Q = 3012
    \item Q = 6012
\end{enumerate}
\end{enumerate}
