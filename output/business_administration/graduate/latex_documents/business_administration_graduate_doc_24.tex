\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item A sofa has a markup of 20\% based on its selling price. Find the percent markup based on the cost.
\begin{enumerate}[label=(\alph*)]
    \item 28\%
    \item 15\%
    \item 22\%
    \item 10\%
    \item 30\%
\end{enumerate}
\item The Century Shoe Company bought a machine for \$1200. Depreciation was charged at the rate of 20\% of the declining balance. Find the book value at the end of 3 years.
\begin{enumerate}[label=(\alph*)]
    \item \$840.00
    \item \$729.60
    \item \$512.00
    \item \$691.20
    \item \$768.00
\end{enumerate}
\item How much interest did Paul pay on his loan of \$1,285 for 2(1 / 2) years, given that the rate of interest was 5(1 / 2) \%?
\begin{enumerate}[label=(\alph*)]
    \item \$200.00
    \item \$176.69
    \item \$165.50
    \item \$150.75
    \item \$135.20
\end{enumerate}
\item The Alexander's monthly telephone bill averaged \$17.05 last year, which was 10\% more than the monthly charge for the previous year. What was the monthly charge the previous year?
\begin{enumerate}[label=(\alph*)]
    \item \$14.95
    \item \$15.95
    \item \$16.00
    \item \$14.50
    \item \$15.05
\end{enumerate}
\item A record sells for \$3.24. There is a 35\% markup based on the selling price. Find the cost of the record.
\begin{enumerate}[label=(\alph*)]
    \item \$1.13
    \item \$3.59
    \item \$2.70
    \item \$2.00
    \item \$2.40
\end{enumerate}
\end{enumerate}
