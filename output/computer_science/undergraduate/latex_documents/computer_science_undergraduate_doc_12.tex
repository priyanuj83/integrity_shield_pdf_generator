\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item In Python 3, what is the following function returns item from the list with max value?
\begin{enumerate}[label=(\alph*)]
    \item cmp(list)
    \item len(list)
    \item max(list)
    \item min(list)
\end{enumerate}
\item In Python 3, b = [11,13,15,17,19,21]; print(b[::2]) outputs what?
\begin{enumerate}[label=(\alph*)]
    \item [19,21]
    \item [11,15]
    \item [11,15,19]
    \item [13,17,21]
\end{enumerate}
\item In Python 3, which of the following operator in python performs exponential (power) calculation on operands?
\begin{enumerate}[label=(\alph*)]
    \item **
    \item //
    \item is
    \item not in
\end{enumerate}
\item Which of the following is the hexadecimal representation of the decimal number 231\_\{10\} ?
\begin{enumerate}[label=(\alph*)]
    \item 17\_\{16\}
    \item E$_{4}$\_\{16\}
    \item E$_{7}$\_\{16\}
    \item F$_{4}$\_\{16\}
\end{enumerate}
\item Which types of functions grow the slowest?
\begin{enumerate}[label=(\alph*)]
    \item O($N^(1$/2))
    \item O($N^(1$/4))
    \item O($N^(1$/N))
    \item O(N)
\end{enumerate}
\end{enumerate}
