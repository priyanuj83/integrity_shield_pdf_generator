\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item In Python 3, which of the following function removes all leading and trailing whitespace in string?
\begin{enumerate}[label=(\alph*)]
    \item replace(old, new [, max])
    \item strip([chars])
    \item swapcase()
    \item title()
\end{enumerate}
\item Which of the following regular expressions is equivalent to (describes the same set of strings as) (a* + b)*(c + d)?
\begin{enumerate}[label=(\alph*)]
    \item a*(c + d)+ b(c + d)
    \item a*(c + d)* + b(c + d)*
    \item a*(c + d)+ b*(c + d)
    \item (a + b)*c +(a + b)*d
\end{enumerate}
\item In Python 3, what is the output of print list[1:3] if list = [ 'abcd', 786 , 2.23, 'john', 70.2 ]?
\begin{enumerate}[label=(\alph*)]
    \item [ 'abcd', 786 , 2.23, 'john', 70.2 ]
    \item abcd
    \item [786, 2.23]
    \item None of the above.
\end{enumerate}
\item What is the output of the statement "a" + "ab" in Python 3?
\begin{enumerate}[label=(\alph*)]
    \item Error
    \item aab
    \item ab
    \item a ab
\end{enumerate}
\item Which of the following best explains what happens when a new device is connected to the Internet?
\begin{enumerate}[label=(\alph*)]
    \item A device driver is assigned to the device.
    \item An Internet Protocol (IP) address is assigned to the device.
    \item A packet number is assigned to the device.
    \item A Web site is assigned to the device.
\end{enumerate}
\end{enumerate}
