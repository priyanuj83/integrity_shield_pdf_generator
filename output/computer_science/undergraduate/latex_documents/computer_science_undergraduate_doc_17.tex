\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item Which types of functions grow the slowest?
\begin{enumerate}[label=(\alph*)]
    \item O(log $e^N$)
    \item O(log N)
    \item O(log log N)
    \item O(N)
\end{enumerate}
\item In Python 3, which of the following function sets the integer starting value used in generating random numbers?
\begin{enumerate}[label=(\alph*)]
    \item choice(seq)
    \item randrange ([start,] stop [,step])
    \item random()
    \item seed([x])
\end{enumerate}
\item Of the following sorting algorithms, which has a running time that is LEAST dependent on the initial ordering of the input?
\begin{enumerate}[label=(\alph*)]
    \item Insertion sort
    \item Quicksort
    \item Merge sort
    \item Selection sort
\end{enumerate}

\end{enumerate}
\item Let x = 1. What is x \textless{}\textless{} 3 in Python 3?

    \item 1
    \item 3
    \item 8
    \item 16
\end{enumerate}
\end{enumerate}
