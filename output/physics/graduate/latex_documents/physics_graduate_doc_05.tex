\section*{Multiple Choice (10 marks)}
\begin{enumerate}[label=\arabic*.]
\item A particle moves with $v=$ const. along the curve $r=k(1+\cos \theta)$ (a cardioid). Find $\ddot{\mathbf{r}} \cdot \mathbf{e}_r=\mathbf{a} \cdot \mathbf{e}_r$.
\begin{enumerate}[label=(\alph*)]
    \item $-\frac{v^2}{k}$
    \item $-\frac{3}{4}$ $\frac{v^2}{k}$
    \item $-\frac{7}{4}$ $\frac{v^2}{k}$
    \item $\frac{1}{4}$ $\frac{v^2}{k}$
    \item $-\frac{1}{4}$ $\frac{v^2}{k}$
\end{enumerate}
\item Find the ratio of the focal lengths of a glass lens in water and in air. The refractive indices of the glass and water are 1.5 and 1.33 respectively.
\begin{enumerate}[label=(\alph*)]
    \item 2.50
    \item 0.89
    \item 2.23
    \item 2.75
    \item 4.5
\end{enumerate}
\item Find the dimension of the parallelepiped of maximum volume circumscribed by a sphere of radius $R$.
\begin{enumerate}[label=(\alph*)]
    \item $\frac{2}{\sqrt{2}}R$
    \item $\frac{3}{\sqrt{2}}R$
    \item $R$
    \item $\sqrt{2}R$
    \item $\frac{1}{\sqrt{3}}R$
\end{enumerate}
\item A uniformly charged conducting sphere of $1.2 \mathrm{~m}$ diameter has surface charge density $8.1 \mu \mathrm{C} / \mathrm{m}^2$. Find the net charge on the sphere.
\begin{enumerate}[label=(\alph*)]
    \item $97 \mu \mathrm{C}$
    \item $37$$\mu \mathrm{C}$
    \item $77 \mu \mathrm{C}$
    \item $87 \mu \mathrm{C}$
    \item $17 \mu \mathrm{C}$
\end{enumerate}
\item An airplane with its nose pointing north with an airspeed of 40 km/h in a 30-km/h crosswind (at right angles) has a groundspeed of
\begin{enumerate}[label=(\alph*)]
    \item 40 km/h
    \item 60 km/h
    \item 50 km/h
    \item 10 km/h
    \item 90 km/h
\end{enumerate}
\end{enumerate}

\section*{True / False (10 marks)}

\begin{enumerate}[label=\arabic*.]

\item True or False: A planet that shrinks to one-tenth its original diameter would cause a person on its surface to weigh 1/10 as much.
\item True or False: In a series circuit, adding more lamps will cause the current from the power source to increase due to higher voltage availability.
\item True or False: The distance between the object and its image in a plane mirror is 200 cm when the object is 100 cm from the mirror.
\item True or False: Satellites are propelled by constant rocket thrust to maintain their orbits around Earth.
\item True or False: During an isothermal expansion, 150 J of heat is added to the gas to compensate for the work done on the surroundings.
\end{enumerate}
