\section*{Multiple Choice (10 marks)}

    \item 0.75 m/sec
    \item 3.5 m/sec
    \item 0.5 m/sec
    \item 1.5 m/sec

    \item 2.0
    \item 2.5

    \item $3.8 \times 10^{11}$
